name: build-news
on:
  schedule:
    - cron: "*/20 * * * *"
  workflow_dispatch:
permissions:
  contents: write
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: "20"
      - run: node scripts/build-news.mjs
      - run: |
          git config user.name "news-bot"
          git config user.email "actions@users.noreply.github.com"
          git add news.json
          git commit -m "chore: refresh news.json" || echo "no change"
          git push
